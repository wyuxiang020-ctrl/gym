type JsonObject = Record<string, unknown>

function isObject(value: unknown): value is JsonObject {
  return typeof value === 'object' && value !== null && !Array.isArray(value)
}

function isFiniteNumber(value: unknown, minimum = 0): value is number {
  return typeof value === 'number' && Number.isFinite(value) && value >= minimum
}

function optionalString(value: unknown): boolean {
  return value === undefined || typeof value === 'string'
}

function optionalStrings(value: unknown): boolean {
  return value === undefined || (Array.isArray(value) && value.every((item) => typeof item === 'string'))
}

function validFoodItem(value: unknown): boolean {
  return (
    isObject(value) &&
    typeof value.name === 'string' &&
    Boolean(value.name.trim()) &&
    isFiniteNumber(value.grams) &&
    isFiniteNumber(value.kcal) &&
    isFiniteNumber(value.protein) &&
    isFiniteNumber(value.carbs) &&
    isFiniteNumber(value.fat) &&
    ['high', 'mid', 'low'].includes(String(value.confidence))
  )
}

function validStrength(value: unknown): boolean {
  return (
    isObject(value) &&
    typeof value.name === 'string' &&
    Boolean(value.name.trim()) &&
    Array.isArray(value.sets) &&
    value.sets.every(
      (set) =>
        isObject(set) &&
        isFiniteNumber(set.weight) &&
        isFiniteNumber(set.reps, 1) &&
        typeof set.done === 'boolean',
    ) &&
    (value.intensity === undefined || ['low', 'mid', 'high'].includes(String(value.intensity))) &&
    optionalString(value.note) &&
    optionalStrings(value.uncertain)
  )
}

function validCardio(value: unknown): boolean {
  return (
    isObject(value) &&
    typeof value.type === 'string' &&
    Boolean(value.type.trim()) &&
    isFiniteNumber(value.minutes, 1) &&
    (value.distance === undefined || value.distance === null || isFiniteNumber(value.distance)) &&
    (value.avgHr === undefined || value.avgHr === null || isFiniteNumber(value.avgHr, 1)) &&
    ['low', 'mid', 'high'].includes(String(value.intensity)) &&
    optionalString(value.note) &&
    optionalStrings(value.uncertain)
  )
}

function invalidResult(value: unknown): never {
  const error = new Error('AI returned JSON with an invalid data structure') as Error & { rawText?: string }
  error.rawText = JSON.stringify(value)
  throw error
}

export function validateWorkoutResult(value: unknown): unknown {
  if (
    !isObject(value) ||
    !Array.isArray(value.strength) ||
    !value.strength.every(validStrength) ||
    !Array.isArray(value.cardio) ||
    !value.cardio.every(validCardio)
  ) {
    return invalidResult(value)
  }
  return value
}

export function validateFoodResult(value: unknown): unknown {
  if (!isObject(value) || !Array.isArray(value.items) || !value.items.every(validFoodItem)) {
    return invalidResult(value)
  }
  return value
}
